	\section*{\textbf{一. 实验名称：氢氘光谱实验}}
	
	\section*{\textbf{二. 实验目的}}
	\begin{enumerate}
		\item 对氢原子光谱巴尔末线系在可见光区域内谱线的测量与分析，学习光栅光谱仪的工作原理和谱线测量的基本技术，学习确定里德伯常量的方法。
		\item 以氘原子光谱为研究对象，研究获得同位素光谱的实验方法、分析方法及其在微观测量中的应用。
	\end{enumerate}

	\section*{\textbf{{三. 实验原理}}}

	
	\begin{enumerate}

	\item \textbf{类氢原子光谱}

	
	氢原子光谱图可以明显地看到有三个谱线系列，一个谱线系列在可见光和近紫外区，称为巴尔末系；一个谱线系列在紫外，为莱曼系；另一个谱线系列在红外，为帕邢系。此外，在长波长方向还有一些不很清楚的线系。每个谱线系都很有规律，间隔向短波方向递减变密变弱。巴尔末发现谱线的波长与谱线的序号n有关，n=1,2,3...，它们之间的关系是：$$\lambda=B\frac{n^2}{n^2-4}$$其中$B=3.6465\times 10^{-7}m$，是里德伯常量。瑞典物理学家里德伯（Johannes Rydberg）将公式改写为以波数$\sigma$表示的形式：
$$\sigma=\frac{1}{\lambda}=\frac{1}{B}\left(\frac{1}{2^2}-\frac{1}{n^2}\right)=\frac{2\pi^2m_ee^4Z^2}{(4\pi\varepsilon_0)^2h^3c(1+\frac{m_e}{m_Z})}\left(\frac{1}{2^2}-\frac{1}{n^2}\right)$$
其中$Z$为原子核的电荷数，$m_e$为电子的质量，$m_Z$为原子核的质量，$\varepsilon_0$为真空介电常数，$h$为普朗克常数，$c$为光速，$e$为电子的电荷量。里德伯常数可写为：
$$R=\frac{2\pi^2m_ee^4Z^2}{(4\pi\varepsilon_0)^2h^3c(1+\frac{m_e}{m_Z})}$$
若$m_Z\rightarrow\infty$，则$R_\infty=\frac{2\pi^2m_ee^4Z^2}{(4\pi\varepsilon_0)^2h^3c}=(10973731.534\pm0.012)m^{-1}$，所以$R_Z=\frac{R_\infty}{(1+\frac{m_e}{m_Z})}$。
因为$\sigma_H=R_H\left(\frac{1}{2^2}-\frac{1}{n^2}\right)$，$\sigma_D=R_D\left(\frac{1}{2^2}-\frac{1}{n^2}\right)$，所以有：$$\Delta\lambda=\lambda_H-\lambda_D=\lambda_H\left(1-\frac{\lambda_D}{\lambda_H}\right)=\lambda_H(1-\frac{\sigma_H}{\sigma_D})=\lambda_H(1-\frac{R_H}{R_D})$$
因为$R_H=\frac{R_\infty}{(1+\frac{m_e}{m_H})}$，$R_D=\frac{R_\infty}{(1+\frac{m_e}{m_D})}$，所以有：$$\frac{R_H}{R_D}=\frac{(1+\frac{m_e}{m_D})}{(1+\frac{m_e}{m_H})}\Leftrightarrow\frac{m_D}{m_H}=\frac{\frac{R_D}{R_H}}{1-\frac{m_H}{m_e}(\frac{R_D}{R_H}-1)}$$

	\item \textbf{实验仪器}

	\begin{enumerate}
		\item 氢氘放电管：当一定的高压加在氢氘放电管的两极上时，管内游离电子受电场作用飞向阳极，碰撞氢氘原子，解离成氢离子和氘原子，激发其能级，回到低能级时发射光辐射。
		\item 光栅光谱仪：利用光栅的衍射特性将不同波长的光分开，光路图如图所示。通过G的旋转，可以使不同波长的光依次通过狭缝S进入光电倍增管进行测量。
		\begin{figure}[H]
			\centering
			\includegraphics[width=10cm]{1.jpg}
			\caption{光栅光谱仪光路图}
		\end{figure}
		\item 光电倍增管：利用光电效应，电流放大元件，将光信号转换为电信号并倍增放大。
	\end{enumerate}

	\end{enumerate}
	
	\section*{\textbf{{四. 实验仪器}}}
	氢氘放电管、汞灯、光栅光谱仪、光电倍增管

	\section*{\textbf{{五. 实验内容}}}

	\begin{enumerate}
		\item 将光谱仪的电压旋钮逆时针旋至最小$\longrightarrow$打开光谱仪电源$\longrightarrow$右键以\textbf{管理员身份运行}“WGD8A”软件, 光谱仪自动复位到200nm 位置,等待复位成功。

		\item \textbf{光谱仪的定标 (Hg灯光谱的测量)}
		\begin{enumerate}
			\item 将 Hg灯安放在光谱仪入射狭缝前,开启电源
			\item 光谱仪上的负高压调至 500V
			\item \textbf{软件设置}
			\begin{enumerate}
				\item 模式:能量; 间隔:0.05nm; 起始波长:350nm; 终止波长:590nm; 最大值:1000; 最小值:0; 采集次数:4; 增益:3;
				\item 点击软件上“单程”图标,开始扫描。
			\end{enumerate}
			\item \textbf{扫描后}:
			\begin{enumerate}
				\item 如Hg灯的谱线峰值高过 1000,适当降低电压值,重新扫描;
				\item 如Hg灯的谱线峰值过低(最高峰$<$700),适当提高电压值,重新扫描;
				\item 其中508nm 附近为二级衍射峰,可视作干扰,不论峰值高低都应人为忽略。
			\end{enumerate}
			\item \textbf{数据记录}
			\begin{enumerate}
				\item 依次记录谱线中每个峰对应的波长(共9个峰)。
				\item 在桌面创建一个Word 文档,保存屏幕截图(图1)。
			\end{enumerate}
		\end{enumerate}

		\item \textbf{HD灯的测量}
		\begin{enumerate}
			\item 光谱仪上的负高压调至800V 左右(最大可调到 1000V)。
			\item 将 HD 灯摆放在光谱仪的入射狭缝前,从 HD 灯后部的小孔望进去,并移动HD灯,找到光线的焦平面(即最细最亮的线),并使其与光谱仪的入射狭缝重合,以便尽可能多的光线进入光谱仪。
			\item \textbf{粗扫 HD 光谱,确定谱线峰值的大概范围}
			\begin{enumerate}
				\item 间隔:0.05nm; 起始波长:350nm; 终止波长:660nm; 单程扫描。
				\item 如扫描后谱线峰值过低,应调整 HD 灯的摆放位置或适当提高电压值,重新扫描,找到3个明显的峰。(此处峰值高过 1000 不影响大概范围的判断)
				\item 截屏并保存至桌面的 Word 文档中(图2)。
			\end{enumerate}
			\item \textbf{细扫 HD 光谱,确定谱线峰值的准确波长}
			\begin{enumerate}
				\item 设定采集间隔为0.01nm,依次在 $n=3\backslash 4\backslash 5$(即 656nm$\backslash$486nm$\backslash$434nm)的3个峰附近$\pm 5$nm的范围内扫描。光谱仪上的负高压可根据谱线的高低而调节。目的将每一个$n$值(主量子数)对应的H、D谱线峰值分离开,共需测出3组($n=3\backslash 4\backslash 5$)HD 双峰。
				\item 如H、D谱线峰值不能分离(20nm范围的谱线只有一个峰值),可微调 HD灯的摆放位置。确实不能分离时,可请老师协助,切不可私自调整狭缝。
				\item 得到H、D双峰后,记录峰值波长(短波长为D,长波长为H)。并截图保存至桌面的 Word 文档中(图3/4/5)。
			\end{enumerate}
		\end{enumerate}

		\item \textbf{数据处理} % Note: The original document uses a main list number '4' here.
		\begin{enumerate} % Using a standard list for (a)-(f)
			\item 拟合修正公式
			\item 修正实验所测的氢波长和氘波长
			\item 计算氢的里德伯常量$R_H$, 并计算平均值
			\item 计算氘的里德伯常量$R_D$, 并计算平均值
			\item 计算氘和氢原子核质量比$\frac{m_D}{m_H}$
		\end{enumerate}

	\end{enumerate}

	\section*{\textbf{{七. 数据处理}}}

	
	\begin{enumerate}
		\item \textbf{拟合修正公式}

	以Hg灯的测量波长为横轴，理论Hg波长为纵轴，进行线性拟合，有拟合曲线如下图所示：

		\begin{figure}[H]
			\centering
			\includegraphics[width=17cm]{2.png}
			\caption{Hg灯标定拟合曲线}
		\end{figure}

	$$\lambda_{Hg}=\alpha+\beta\lambda_{Hg}^0=2.9271 + 0.99315 \lambda_{Hg}^0
	$$

	\item \textbf{修正后的HD灯的波长}

	$$\lambda_{H,3}=\alpha+\beta\lambda_{H,3}^0=2.9271 + 0.99315 \times 658.58 = 656.99\mathrm{nm}$$

	$$\lambda_{D,3}=\alpha+\beta\lambda_{D,3}^0=2.9271 + 0.99315 \times 658.40 = 656.82\mathrm{nm}$$

	$$\lambda_{H,4}=\alpha+\beta\lambda_{H,4}^0=2.9271 + 0.99315 \times 486.45 = 486.04\mathrm{nm}$$

	$$\lambda_{D,4}=\alpha+\beta\lambda_{D,4}^0=2.9271 + 0.99315 \times 486.32 = 485.91\mathrm{nm}$$

	$$\lambda_{H,5}=\alpha+\beta\lambda_{H,5}^0=2.9271 + 0.99315 \times 434.05 = 434.00\mathrm{nm}$$

	$$\lambda_{D,5}=\alpha+\beta\lambda_{D,5}^0=2.9271 + 0.99315 \times 433.93 = 433.88\mathrm{nm}$$

	\item \textbf{计算$R_H$}

	$$R_{H,3}=\frac{1}{\lambda_{H,3}(\frac{1}{2^2}-\frac{1}{3^2})}=
	\frac{1}{ 656.99 \mathrm{nm}\times (\frac{1}{2^2}-\frac{1}{ 3 ^2})}= 10961381.48 \mathrm{m}^{-1}$$

	$$R_{H,4}=\frac{1}{\lambda_{H,4}(\frac{1}{2^2}-\frac{1}{4^2})}=
	\frac{1}{ 486.04 \mathrm{nm}\times (\frac{1}{2^2}-\frac{1}{ 4 ^2})}= 10971272.89 \mathrm{m}^{-1}$$

	$$R_{H,5}=\frac{1}{\lambda_{H,5}(\frac{1}{2^2}-\frac{1}{5^2})}=
	\frac{1}{ 434.00 \mathrm{nm}\times (\frac{1}{2^2}-\frac{1}{ 5 ^2})}= 10971351.28 \mathrm{m}^{-1}$$

	$$\bar R_H=\frac{R_{H,3}+R_{H,4}+R_{H,5}}{3}=
	\frac{10961381.48 +10971272.89 +10971351.28 }{3}=10968001.89 \mathrm{m}^{-1}$$

	\item \textbf{计算$R_D$}

	$$R_{D,3}=\frac{1}{\lambda_{D,3}(\frac{1}{2^2}-\frac{1}{3^2})}=
	\frac{1}{ 656.82 \mathrm{nm}\times (\frac{1}{2^2}-\frac{1}{ 3 ^2})}= 10964542.58 \mathrm{m}^{-1}$$

	$$R_{D,4}=\frac{1}{\lambda_{D,4}(\frac{1}{2^2}-\frac{1}{4^2})}=
	\frac{1}{ 485.91 \mathrm{nm}\times (\frac{1}{2^2}-\frac{1}{ 4 ^2})}=10973972.90 \mathrm{m}^{-1}$$

	$$R_{D,5}=\frac{1}{\lambda_{D,5}(\frac{1}{2^2}-\frac{1}{5^2})}=
	\frac{1}{ 433.88 \mathrm{nm}\times (\frac{1}{2^2}-\frac{1}{ 5 ^2})}= 10974375.42 \mathrm{m}^{-1}$$

	$$\bar R_D=\frac{R_{D,3}+R_{D,4}+R_{D,5}}{3}=
	\frac{10964542.58 +10973972.90 +10974375.42 }{3}=10970963.64 \mathrm{m}^{-1}$$

	\item \textbf{计算氘和氢原子核质量比}

	$$\frac{m_D}{m_H}=\frac{\frac{R_D}{R_H}}{1-\frac{m_H}{m_e}(\frac{R_D}{R_H}-1)}=\frac{\frac{ 10968001.89 }{ 10970963.64 }}{1-\frac{ 1.674\times10^{-27} }{ 9.109\times10^{-31} }(\frac{ 10970963.64 }{ 10968001.89 }-1)}= 1.979$$。

	\end{enumerate}

	
	\section*{\textbf{八. 误差分析}}

	\begin{enumerate}
		\item 光栅光谱仪本身存在一定的分辨率限制，可能导致谱线测量的精度不足；
		\item 氢氘放电管的稳定性可能影响光谱的强度和清晰度，进而影响测量结果；
		\item 光电倍增管的灵敏度和响应时间可能引入测量误差；
		\item 实验环境中的光干扰可能影响测量的准确性。
	\end{enumerate}

	\section*{\textbf{{十. 实验结论}}}

	本实验使用光栅光谱仪测量氢氘原子光谱，并使用汞灯标定，测得氢的里德伯系数为$R_H=10968001.89 \mathrm{m}^{-1}$，氘的里德伯系数为$R_D=10970963.64 \mathrm{m}^{-1}$，计算得同位素质量比为$\frac{m_D}{m_H}=1.979$，与理论值基本相符。
